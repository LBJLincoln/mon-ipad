# ==========================================================================
# Docling Continuous Ingestion — Codespace Crontab
# ==========================================================================
# Install: crontab codespace/crontab.txt
# Verify:  crontab -l
# Remove:  crontab -r
#
# Paths assume Codespace working directory: /workspaces/mon-ipad
# Adjust if running from a different location (e.g., /home/termius/mon-ipad)
# ==========================================================================

# Environment: use bash shell for source command
SHELL=/bin/bash

# Docling continuous ingestion — every 30 minutes
# Discovers new expert PDFs, processes with Docling, ingests to Pinecone/Supabase/Neo4j
*/30 * * * * cd /workspaces/mon-ipad && source .env.local && python3 codespace/docling-cron.py --max-pdfs 10 >> codespace/logs/cron.log 2>&1

# Push processed URLs and logs to GitHub — every 2 hours
# Only commits if there are changes in codespace/data/ or codespace/logs/
0 */2 * * * cd /workspaces/mon-ipad && git add codespace/data/ codespace/logs/docling-cron.jsonl && git diff --cached --quiet || (git commit -m "auto: docling cron — $(date -u +\%Y-\%m-\%dT\%H:\%MZ)" && git push origin main) 2>/dev/null

# Cleanup old temp files — daily at 3 AM
0 3 * * * find /workspaces/mon-ipad/codespace/tmp -name "*.pdf" -mtime +1 -delete 2>/dev/null

# Rotate cron.log — weekly on Sunday at 4 AM (keep last 10K lines)
0 4 * * 0 cd /workspaces/mon-ipad/codespace/logs && tail -10000 cron.log > cron.log.tmp && mv cron.log.tmp cron.log 2>/dev/null
